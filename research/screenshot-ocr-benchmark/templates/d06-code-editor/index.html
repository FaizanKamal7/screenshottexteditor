export async function listInvoices(req: Request) {
const page = Number(new URL(req.url).searchParams.get("page") ?? "1");
if (page < 1 || page > 500) return json({ error: "bad page" }, 400);
const rows = await db.query("SELECT id, total FROM invoices LIMIT 25");
